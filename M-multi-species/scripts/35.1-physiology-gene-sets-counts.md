``` r
apul_counts <- read.csv("../../D-Apul/output/02.20-D-Apul-RNAseq-alignment-HiSat2/apul-gene_count_matrix.csv")

ptua_counts <- read.csv("../../F-Ptua/output/02.20-F-Ptua-RNAseq-alignment-HiSat2/ptua-gene_count_matrix.csv")
```

Read in gene sets

``` r
ant_genes <- read.csv("../output/35-physiology-gene-sets/antioxidant_geneset_expanded.csv")
grow_genes <- read.csv("../output/35-physiology-gene-sets/growth_geneset_expanded.csv")
resp_genes <- read.csv("../output/35-physiology-gene-sets/respiration_geneset_expanded.csv")
sym_genes <- read.csv("../output/35-physiology-gene-sets/symbiosis_geneset_expanded.csv")

# Remove -TXX suffix on Apul gene names
ant_genes$apul <- sub("-.*", "", ant_genes$apul)
grow_genes$apul <- sub("-.*", "", grow_genes$apul)
resp_genes$apul <- sub("-.*", "", resp_genes$apul)
sym_genes$apul <- sub("-.*", "", sym_genes$apul)

# Add gene- prefix on Peve gene names
ant_genes$peve <- paste0("gene-", ant_genes$peve)
grow_genes$peve <- paste0("gene-", grow_genes$peve)
resp_genes$peve <- paste0("gene-", resp_genes$peve)
sym_genes$peve <- paste0("gene-", sym_genes$peve)

# Add gene- prefix on Peve gene names
ant_genes$ptua <- paste0("gene-", ant_genes$ptua)
grow_genes$ptua <- paste0("gene-", grow_genes$ptua)
resp_genes$ptua <- paste0("gene-", resp_genes$ptua)
sym_genes$ptua <- paste0("gene-", sym_genes$ptua)
```

Filter

``` r
# Apul
apul_counts_ant <- apul_counts[apul_counts$gene_id %in% ant_genes$apul,]
apul_counts_grow <- apul_counts[apul_counts$gene_id %in% grow_genes$apul,]
apul_counts_resp <- apul_counts[apul_counts$gene_id %in% resp_genes$apul,]
apul_counts_sym <- apul_counts[apul_counts$gene_id %in% sym_genes$apul,]

# Peve
peve_counts_ant <- peve_counts[peve_counts$gene_id %in% ant_genes$peve,]
peve_counts_grow <- peve_counts[peve_counts$gene_id %in% grow_genes$peve,]
peve_counts_resp <- peve_counts[peve_counts$gene_id %in% resp_genes$peve,]
peve_counts_sym <- peve_counts[peve_counts$gene_id %in% sym_genes$peve,]

# Ptua
ptua_counts_ant <- ptua_counts[ptua_counts$gene_id %in% ant_genes$ptua,]
ptua_counts_grow <- ptua_counts[ptua_counts$gene_id %in% grow_genes$ptua,]
ptua_counts_resp <- ptua_counts[ptua_counts$gene_id %in% resp_genes$ptua,]
ptua_counts_sym <- ptua_counts[ptua_counts$gene_id %in% sym_genes$ptua,]
```
